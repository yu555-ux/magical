变量格式总览:
  世界信息:
    梦境定位:
      type: |-
        { 即将入睡: boolean; 位于梦境: boolean; } 
    现实, 梦境存档:
      type: |-
        { 时间: format: YYYY年MM月DD日-星期X-HH:MM; 地点, 天气 ; }

  主角:
    年龄:
      type: number
    评级:
      value: "'微尘'|'聚砂'|'凝石'|'磐岩'|'撼山'|'摧城'|'覆国'|'夷地'|'灭世'"
    性器:
      type: |-
        { 疲软长度: number; 勃起长度: number; }
    资源:
      type: |-
        { 超凡资源: { 蝶烬: number; 尸气: number; }; 金钱: { 单位: 元; 数值: number; }; }
    身体属性.${生命|体力|能量|SAN}:
      type: |-
        { 当前: number; 上限: number; }
    基础属性.${力量|体质|精神|敏捷}:
      type: number; range: 1~100
    特殊属性.${幸运|魅力}:
      type: number; range: 1~100
    状态:
      type: |-
        { [状态名: string]: { 描述: string; 持续时间?: string; }; }
    技能:
      type: |-
        {
          [技能名: string]: {
            等级: "'微尘'|'聚砂'|'凝石'|'磐岩'|'撼山'|'摧城'|'覆国'|'夷地'|'灭世'";
            描述, 使用要求: string;
            消耗能量: number;
            副作用: { [副作用名: string]: string; };
            熟练度: number; range: 0~999
            分支: { [分支名: string]: { 描述: string; 效果: string; } };
          };
        }
    持有物品:
      灵宝:
        type: |-
          {
            [灵宝名: string]: {
              等级: "'微末'|'凶煞'|'祸城'|'倾国'|'绝域'|'灭世'";
              描述: string; 数量: number;
              效果: { [效果名: string]: string; };
            };
          }
      诡物:
        type: |-
          {
            [诡物名: string]: {
              等级: "'微末'|'凶煞'|'祸城'|'倾国'|'绝域'|'灭世'";
              描述: string; 数量: number;
              效果: { [效果名: string]: string; };
              规则: { [规则名: string]: string; }; // 仅诡物
              副作用: { [副作用名: string]: string; }; // 仅诡物
            };
          }
      物品:
        type: |-
          {
            [物品名: string]: {
              等级?: string; 数量: number; 描述: string;
              效果?: { [效果名: string]: string; };
            };
          }
    社交:
      type: |-
        { [角色名: string]: { 关系: string; 社交圈: { [角色名: string]: string; }; }; }

  仓库: 结构同“主角.持有物品”

  主要人物·女性.${异人|普通人}.${角色名}:
    检索词, 身份, 当前位置, 当前行动, 当前想法:
    梦境NPC:
      type: boolean
    年龄:
      type: number
    评级:
      value: "'微尘'|'聚砂'|'凝石'|'磐岩'|'撼山'|'摧城'|'覆国'|'夷地'|'灭世'" // 仅异人
    好感值:
      type: number; range: -200~200
    堕落值:
      type: number; range: 0~500
    性欲值:
      type: number; range: 0~100
    社交圈:
      type: |-
        { [角色名: string]: string; }
    状态:
      type: |-
        { [状态名: string]: { 描述: string; 持续时间?: string; }; }
    身体属性:
      type: |-
        {
          生命: { 当前: number; 上限: number; };
          能量?: { 当前: number; 上限: number; }; // 仅异人
          SAN:  { 当前: number; 上限: number; };
        }
    基础属性.${力量|体质|精神|敏捷}:
      type: number; range: 1~100
    特殊属性.${幸运|魅力}:
      type: number; range: 1~100
    着装.${上衣|下衣|内衣|袜子|鞋子}:
      type: |-
        { 名称: string; 描述: string; }
    身体开发.${嘴巴|胸部|小穴|屁穴}:
      type: |-
        { 描述: string; 使用次数: number; }
    技能:
      type: |-
        {
          [技能名: string]: {
            等级: "'微尘'|'聚砂'|'凝石'|'磐岩'|'撼山'|'摧城'|'覆国'|'夷地'|'灭世'";
            描述, 使用要求: string;
            副作用: { [副作用名: string]: string; };
            熟练度: number; range: 0~999
            分支: { [分支名: string]: { 描述: string; 效果: string; } };
          };
        } // 仅异人
    所持物品:
      note: 结构同 主角.持有物品，无 梦境物品 字段 // 仅异人
    子宫:
      type: |-
        {
          宫内精液: {
            总量: number;  // 所有来源容量之和，代码自动同步
            来源列表: Array<{
              来源: string;
              容量: number;  // ml, 该来源当前剩余量（衰减后）
              注入时间: string;  // format: YYYY年MM月DD日-HH:MM
            }>;
          };
          生理周期: {
            上次经期日: format: YYYY年MM月DD日; 周期天数: number; 经期长度: number;
          };
          生育记录: Array<{ 生产日期: string; 父方: string; 孩子: string; }>; 
        }

  主要人物·男性.${异人|普通人}.${角色名}:
    检索词, 身份, 当前位置, 当前行动, 当前想法:
    梦境NPC:
      type: boolean
    年龄:
      type: number
    评级:
      value: "'微尘'|'聚砂'|'凝石'|'磐岩'|'撼山'|'摧城'|'覆国'|'夷地'|'灭世'" // 仅异人
    友善值:
      type: number; range: -200~200
    社交圈:
      type: |-
        { [角色名: string]: string; }
    状态:
      type: |-
        { [状态名: string]: { 描述: string; 持续时间?: string; }; }
    身体属性:
      type: |-
        {
          生命: { 当前: number; 上限: number; };
          能量?: { 当前: number; 上限: number; }; // 仅异人
          SAN:  { 当前: number; 上限: number; };
        }
    基础属性.${力量|体质|精神|敏捷}:
      type: number; range: 1~100
    特殊属性.${幸运|魅力}:
      type: number; range: 1~100
    技能:
      type: |-
        {
          [技能名: string]: {
            等级: "'微尘'|'聚砂'|'凝石'|'磐岩'|'撼山'|'摧城'|'覆国'|'夷地'|'灭世'";
            描述, 使用要求: string;
            副作用: { [副作用名: string]: string; };
            熟练度: number; range: 0~999
            分支: { [分支名: string]: { 描述: string; 效果: string; } };
          };
        } // 仅异人
    所持物品:
      note: 结构同 主角.持有物品，无 梦境物品 字段 // 仅异人

  地图:
    ${地点名}:
      type: |-
        {
          检索词: string[];
          方位?: { X: [number, number]; Y: [number, number]; Z: [number, number]; }; // 世界级无需
          现实, 梦境: {
            描述: string;
            地点细节: {
              信息: string[];
              异常: {
                [异常名: string]: {
                  评级, 描述: string;
                  具现进度?: number; range: 0~100 // 仅梦境
                  特性?: { [特性名: string]: { 描述: string; 效果: string[]; } };
                };
              };
            };
          };
          子地图: { [子地点名: string]: { ...同上 }; };
        }

  特殊玩法:
    柳三娘商店:
      交易开放:
        type: boolean

  柳三娘商店:
    商品:
      type: |-
        {
          [商品名: string]: {
            分类: "'灵宝'|'诡物'|'物品'"; 价格: number; 等级, 描述: string;
            效果: { [效果名: string]: string; };
            规则?: { [规则名: string]: string; };
            副作用?: { [副作用名: string]: string; };
            库存: number;
          };
        }
